import { Lesson, Step, Scene } from './lesson-types';
import gsap from 'gsap';
import { useState, useCallback, useMemo, useEffect } from 'react';

export class LessonEngine {
  private lesson: Lesson;
  private currentSceneIndex: number = 0;
  private currentStepIndex: number = 0;
  private masterTimeline: gsap.core.Timeline;
  private onStateChange?: (state: EngineState) => void;

  constructor(lesson: Lesson, onStateChange?: (state: EngineState) => void) {
    this.lesson = lesson;
    this.onStateChange = onStateChange;
    this.masterTimeline = gsap.timeline({
      onUpdate: () => this.reportState(),
      onComplete: () => this.onPlaybackComplete()
    });
  }

  public static useLessonEngine(lesson: Lesson) {
    const [state, setState] = useState<EngineState>(() => ({
       currentSceneIndex: 0,
       currentStepIndex: 0,
       isPlaying: false,
       progress: 0,
       currentStep: lesson.scenes[0].steps[0]
    }));

    const engine = useMemo(() => new LessonEngine(lesson, setState), [lesson]);
    
    return { engine, state };
  }

  private reportState() {
    if (this.onStateChange) {
      this.onStateChange({
        currentSceneIndex: this.currentSceneIndex,
        currentStepIndex: this.currentStepIndex,
        isPlaying: this.masterTimeline.isActive(),
        progress: this.masterTimeline.progress(),
        currentStep: this.getCurrentStep()
      });
    }
  }

  private onPlaybackComplete() {
    this.reportState();
  }

  public getCurrentStep(): Step {
    return this.lesson.scenes[this.currentSceneIndex].steps[this.currentStepIndex];
  }

  public play() {
    this.masterTimeline.play();
  }

  public pause() {
    this.masterTimeline.pause();
  }

  public nextStep() {
    const scene = this.lesson.scenes[this.currentSceneIndex];
    if (this.currentStepIndex < scene.steps.length - 1) {
      this.currentStepIndex++;
    } else if (this.currentSceneIndex < this.lesson.scenes.length - 1) {
      this.currentSceneIndex++;
      this.currentStepIndex = 0;
    }
    this.reportState();
  }

  public prevStep() {
    if (this.currentStepIndex > 0) {
      this.currentStepIndex--;
    } else if (this.currentSceneIndex > 0) {
      this.currentSceneIndex--;
      this.currentStepIndex = this.lesson.scenes[this.currentSceneIndex].steps.length - 1;
    }
    this.reportState();
  }

  public seekTo(progress: number) {
    this.masterTimeline.progress(progress);
    this.reportState();
  }
}

export interface EngineState {
  currentSceneIndex: number;
  currentStepIndex: number;
  isPlaying: boolean;
  progress: number;
  currentStep: Step;
}
